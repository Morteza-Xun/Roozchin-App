import flet as ft
import json
import os
import multiprocessing
import threading
import queue
import asyncio
from datetime import datetime

try:
    import jdatetime
    HAS_JDATETIME = True
except ImportError:
    HAS_JDATETIME = False

try:
    import pystray
    from PIL import Image
    HAS_PYSTRAY = True
except ImportError:
    HAS_PYSTRAY = False

DATA_FILE = "days.json"

CATEGORIES = ["تحصیل", "مهارت", "ورزش", "خانواده", "سرگرمی"]

CATEGORY_ICONS = {
    "تحصیل": ft.Icons.SCHOOL,
    "مهارت": ft.Icons.PSYCHOLOGY,
    "ورزش": ft.Icons.FITNESS_CENTER,
    "خانواده": ft.Icons.FAMILY_RESTROOM,
    "سرگرمی": ft.Icons.SPORTS_ESPORTS,
}

CATEGORY_COLORS = {
    "تحصیل": "#FF6B9D",
    "مهارت": "#A29BFE",
    "ورزش": "#00B894",
    "خانواده": "#FDCB6E",
    "سرگرمی": "#74B9FF",
}


def to_persian_digits(s):
    return str(s).translate(str.maketrans("0123456789", "۰۱۲۳۴۵۶۷۸۹"))


def today_jalali():
    if HAS_JDATETIME:
        return jdatetime.datetime.now().strftime("%Y/%m/%d")
    return datetime.now().strftime("%Y/%m/%d")


def format_date_long(date_str):
    if not HAS_JDATETIME:
        return date_str
    try:
        d = jdatetime.datetime.strptime(date_str, "%Y/%m/%d")
        weekdays = {0: 'شنبه', 1: 'یک‌شنبه', 2: 'دوشنبه', 3: 'سه‌شنبه', 4: 'چهارشنبه', 5: 'پنج‌شنبه', 6: 'جمعه'}
        months = ['فروردین', 'اردیبهشت', 'خرداد', 'تیر', 'مرداد', 'شهریور', 'مهر', 'آبان', 'آذر', 'دی', 'بهمن', 'اسفند']
        weekday_fa = weekdays[d.weekday()]
        month_fa = months[d.month - 1]
        return to_persian_digits(f"{weekday_fa} ، {d.day} {month_fa} {d.year}")
    except Exception:
        return date_str


def get_theme_colors(theme_id):
    if theme_id == 0:
        return {"bg": "#F5F5F5", "gradient": None, "text": "#111111", "text_sub": "#555555", "glass_bg": "#FFFFFF", "glass_border": "#E0E0E0", "icon": "#111111", "widget_bg": "#EEFFFFFF"}
    elif theme_id == 1:
        return {"bg": "#000000", "gradient": None, "text": "#FFFFFF", "text_sub": "#AAAAAA", "glass_bg": "#151515", "glass_border": "#333333", "icon": "#FFFFFF", "widget_bg": "#EE151515"}
    else:
        return {"bg": "#0F0C29", "gradient": ["#0F0C29", "#302B63", "#24243E"], "text": "#FFFFFF", "text_sub": "#AAAADD", "glass_bg": "#22FFFFFF", "glass_border": "#33FFFFFF", "icon": "#FFFFFF", "widget_bg": "#DD1E1E2E"}


def load_data():
    if not os.path.exists(DATA_FILE):
        return {"days": []}
    try:
        with open(DATA_FILE, "r", encoding="utf-8") as f:
            data = json.load(f)
            for day in data.get("days", []):
                if "categories" in day:
                    old_cats = day.pop("categories")
                    day["custom_categories"] = []
                    for cat_name, tasks in old_cats.items():
                        if not tasks:
                            continue
                        ctype = cat_name if cat_name in CATEGORIES else "شخصی"
                        if ctype == "شخصی":
                            ctype = "تحصیل"
                        day["custom_categories"].append({
                            "id": f"{cat_name}_{day['date']}",
                            "name": cat_name,
                            "type": ctype,
                            "tasks": tasks,
                            "expanded": True
                        })
                if "lesson" not in day:
                    day["lesson"] = ""
                if "custom_categories" not in day:
                    day["custom_categories"] = []
                for cat in day.get("custom_categories", []):
                    if "expanded" not in cat:
                        cat["expanded"] = True
            return data
    except Exception:
        return {"days": []}


def save_data(data):
    with open(DATA_FILE, "w", encoding="utf-8") as f:
        json.dump(data, f, ensure_ascii=False, indent=2)


# ---------- Floating Widget Process ----------
def show_floating_widget(data_file, theme_id, target_date, q_show_main):
    """این تابع در یک پروسه جداگانه اجرا می‌شود"""
    def load_widget_data():
        if not os.path.exists(data_file):
            return {"days": []}
        try:
            with open(data_file, "r", encoding="utf-8") as f:
                return json.load(f)
        except Exception:
            return {"days": []}

    data = load_widget_data()
    day = next((d for d in data["days"] if d["date"] == target_date), None)
    colors = get_theme_colors(theme_id)

    def widget_main(page: ft.Page):
        page.title = f"روزچین - {format_date_long(target_date)}"
        page.rtl = True
        page.padding = 0
        page.window.resizable = False
        page.window.maximizable = False
        page.window.frameless = True
        page.window.always_on_top = True
        page.window.width = 320
        page.window.height = 380
        page.bgcolor = ft.Colors.TRANSPARENT
        page.window.bgcolor = ft.Colors.TRANSPARENT

        page.fonts = {
            "Vazir": "Vazir-Medium.ttf",
            "Vazir Bold": "Vazir-Bold.ttf",
        }
        page.theme = ft.Theme(font_family="Vazir")
        page.dark_theme = ft.Theme(font_family="Vazir")

        def close_widget(e):
            os._exit(0)

        def go_back(e):
            q_show_main.put("show_main")
            os._exit(0)

        def refresh(e=None):
            d = load_widget_data()
            d_day = next((x for x in d["days"] if x["date"] == target_date), None)
            task_col.controls.clear()
            task_col.controls.extend(build_tasks(d_day))
            page.update()

        def build_tasks(d_day):
            if not d_day or not d_day.get("custom_categories"):
                return [ft.Text("کاری ثبت نشده", color=colors["text_sub"], font_family="Vazir")]
            
            all_tasks = []
            for cat in d_day["custom_categories"]:
                for task in cat.get("tasks", []):
                    all_tasks.append((cat["id"], task))
            
            if not all_tasks:
                return [ft.Text("کاری ثبت نشده", color=colors["text_sub"], font_family="Vazir")]

            controls = []
            for cat_id, task in all_tasks[:10]: # نمایش حداکثر ۱۰ کار برای جلوگیری از شلوغی
                is_done = task.get("done", False)
                
                def make_handler(c_id, t_text):
                    def handler(e):
                        val = e.control.value
                        fresh_data = load_widget_data()
                        for day_item in fresh_data["days"]:
                            if day_item["date"] == target_date:
                                for c in day_item["custom_categories"]:
                                    if c["id"] == c_id:
                                        for t in c["tasks"]:
                                            if t["text"] == t_text:
                                                t["done"] = val
                        
                        with open(data_file, "w", encoding="utf-8") as f:
                            json.dump(fresh_data, f, ensure_ascii=False, indent=2)
                        
                        refresh()
                    return handler

                controls.append(
                    ft.Container(
                        content=ft.Row([
                            ft.Checkbox(
                                value=is_done, 
                                on_change=make_handler(cat_id, task["text"]),
                                fill_color=ft.Colors.CYAN_300 if theme_id != 0 else ft.Colors.BLUE
                            ),
                            ft.Text(
                                task["text"], 
                                size=14, 
                                color=colors["text_sub"] if is_done else colors["text"], 
                                expand=True, 
                                selectable=True, 
                                font_family="Vazir",
                                style=ft.TextStyle(
                                    decoration=ft.TextDecoration.LINE_THROUGH if is_done else None
                                )
                            ),
                        ], spacing=4),
                        padding=ft.Padding.symmetric(vertical=2),
                    )
                )
            return controls

        task_col = ft.Column(build_tasks(day), spacing=2, scroll=ft.ScrollMode.AUTO)

        page.add(
            ft.Container(
                content=ft.Column([
                    ft.WindowDragArea(
                        ft.Container(
                            content=ft.Row([
                                ft.Icon(ft.Icons.DRAG_INDICATOR, color=colors["text_sub"], size=20),
                                ft.Text("روزچین", font_family="Vazir Bold", size=16, color=colors["text"], expand=True),
                                ft.IconButton(ft.Icons.REFRESH, icon_size=16, icon_color=colors["text_sub"], on_click=refresh, tooltip="بروزرسانی"),
                                ft.IconButton(ft.Icons.ARROW_BACK, icon_size=18, icon_color=colors["text_sub"], on_click=go_back, tooltip="بازگشت به برنامه"),
                                ft.IconButton(ft.Icons.CLOSE, icon_size=18, icon_color=ft.Colors.RED_400, on_click=close_widget, tooltip="بستن"),
                            ]),
                            padding=ft.Padding.only(left=10, right=5, top=5, bottom=5),
                            bgcolor="#00000000",
                        )
                    ),
                    ft.Container(
                        content=ft.Text(format_date_long(target_date), font_family="Vazir", size=13, color=colors["text_sub"]),
                        padding=ft.Padding.symmetric(horizontal=12, vertical=6),
                    ),
                    ft.Divider(height=1, color=colors["glass_border"]),
                    ft.Container(
                        content=task_col,
                        padding=ft.Padding.all(12),
                        expand=True,
                    ),
                ], spacing=0, expand=True),
                bgcolor=colors["widget_bg"],
                border_radius=16,
                border=ft.Border.all(1, colors["glass_border"]),
                padding=0,
                expand=True,
            )
        )

    ft.run(widget_main, assets_dir="assets")


# ---------- Main App ----------
def main(page: ft.Page):
    page.title = "روزچین"
    page.rtl = True
    page.padding = ft.Padding.all(0)

    # تنظیمات پنجره
    page.window.resizable = True
    page.window.maximizable = True
    page.window.min_width = 360
    page.window.min_height = 520
    page.window.width = 900
    page.window.height = 720
    page.window.icon = r"C:\planner\assets\logo.ico" 

    # ---------- فونت وزیرمتن ----------
    page.fonts = {
        "Vazir": "Vazir-Medium.ttf",
        "Vazir Bold": "Vazir-Bold.ttf",
    }
    page.theme = ft.Theme(font_family="Vazir")
    page.dark_theme = ft.Theme(font_family="Vazir")

    data = load_data()
    state = {"view": "home", "current_date": None, "new_cat_type": "تحصیل", "theme": 2}
    
    # صف ارتباطی سراسری برای بازگرداندن برنامه اصلی
    q_show_main = multiprocessing.Queue()

    root = ft.Container(expand=True)
    page.add(root)

    # --- Theme Helpers ---
    def apply_theme():
        c = get_theme_colors(state["theme"])
        page.bgcolor = c["bg"]
        page.theme_mode = ft.ThemeMode.LIGHT if state["theme"] == 0 else ft.ThemeMode.DARK
        page.update()

    def cycle_theme(e):
        state["theme"] = (state["theme"] + 1) % 3
        apply_theme()
        if state["view"] == "home":
            _render_home()
        elif state["view"] == "editor":
            _render_editor()
        elif state["view"] == "history":
            _render_history()

    apply_theme()

    # --- Async Listener for Widget Return ---
    async def listen_for_return():
        while True:
            try:
                msg = q_show_main.get_nowait()
                if msg == "show_main":
                    # بارگذاری مجدد داده‌ها برای اعمال تغییرات ویجت
                    nonlocal data
                    data = load_data()
                    
                    if state["view"] == "home":
                        _render_home()
                    elif state["view"] == "editor":
                        _render_editor()
                    elif state["view"] == "history":
                        _render_history()

                    page.window.visible = True
                    page.window.to_front()
                    page.update()
            except queue.Empty:
                pass
            await asyncio.sleep(0.5)

    page.run_task(listen_for_return)

    # --- Visual helpers ---
    def background():
        c = get_theme_colors(state["theme"])
        if c["gradient"]:
            return ft.Container(expand=True, gradient=ft.LinearGradient(begin=ft.Alignment.TOP_LEFT, end=ft.Alignment.BOTTOM_RIGHT, colors=c["gradient"]))
        return ft.Container(expand=True, bgcolor=c["bg"])

    def glass(content, padding=16, radius=18):
        c = get_theme_colors(state["theme"])
        return ft.Container(content=content, padding=ft.Padding.all(padding), border_radius=radius, bgcolor=c["glass_bg"], border=ft.Border.all(1, c["glass_border"]), expand=True)

    def make_button(text, icon, on_click, color):
        c = get_theme_colors(state["theme"])
        def on_hover(e):
            e.control.bgcolor = c["glass_border"] if e.data == "true" else c["glass_bg"]
            e.control.update()
        return ft.Container(
            content=ft.Row([ft.Icon(icon, color=color, size=28), ft.Text(text, font_family="Vazir Bold", size=19, weight=ft.FontWeight.BOLD, color=c["text"], selectable=True)], alignment=ft.MainAxisAlignment.CENTER, spacing=12),
            padding=ft.Padding.symmetric(horizontal=24, vertical=18),
            border_radius=20, bgcolor=c["glass_bg"], border=ft.Border.all(1, c["glass_border"]),
            on_click=on_click, on_hover=on_hover, expand=True,
        )

    def get_top_bar():
        c = get_theme_colors(state["theme"])
        return ft.Row([
            ft.IconButton(icon=ft.Icons.BRIGHTNESS_6, icon_color=c["icon"], icon_size=28, tooltip="تغییر تم", on_click=cycle_theme)
        ], alignment=ft.MainAxisAlignment.END)

    # --- Responsive wrapper ---
    def show_view(view_content):
        inner = ft.Container(content=view_content, expand=True)
        root.content = ft.Stack([
            background(),
            ft.Container(
                content=ft.Column([inner], horizontal_alignment=ft.CrossAxisAlignment.CENTER, scroll=ft.ScrollMode.AUTO, expand=True),
                padding=ft.Padding.all(24), expand=True,
            ),
        ], expand=True)
        page.update()

    # --- Widget Launcher Logic ---
    def launch_widget_for_date(target_date):
        page.window.visible = False
        p = multiprocessing.Process(
            target=show_floating_widget, 
            args=(DATA_FILE, state["theme"], target_date, q_show_main), 
            daemon=True
        )
        p.start()

        if HAS_PYSTRAY:
            def on_tray_click(icon, item):
                if item.text == "نمایش برنامه اصلی":
                    page.window.visible = True
                    page.window.to_front()
                    page.update()
                    icon.stop()
                elif item.text == "خروج":
                    icon.stop()
                    os._exit(0)

            icon_path = os.path.join(os.path.dirname(__file__), "assets", "logo.ico")
            try:
                image = Image.open(icon_path)
            except Exception:
                image = Image.new('RGB', (64, 64), color='purple')

            menu = pystray.Menu(
                pystray.MenuItem("نمایش برنامه اصلی", on_tray_click, default=True),
                pystray.MenuItem("خروج", on_tray_click),
            )
            icon = pystray.Icon("roozchin", image, "روزچین", menu)
            threading.Thread(target=icon.run, daemon=True).start()

    # --- Home ---
    def _render_home():
        c = get_theme_colors(state["theme"])
        today = today_jalali()
        today_day = next((d for d in data["days"] if d["date"] == today), None)

        progress_text = ""
        if today_day:
            total = sum(len(cat["tasks"]) for cat in today_day.get("custom_categories", []))
            done = sum(1 for cat in today_day.get("custom_categories", []) for t in cat["tasks"] if t.get("done"))
            progress_text = to_persian_digits(f"امروز: {done} از {total} کار انجام شده")

        def new_day(e):
            t = today_jalali()
            day = next((d for d in data["days"] if d["date"] == t), None)
            if not day:
                day = {"date": t, "quote": "", "lesson": "", "custom_categories": [], "created": datetime.now().isoformat()}
                data["days"].append(day)
                save_data(data)
            state["current_date"] = t
            show_editor()

        view = ft.Column([
            get_top_bar(),
            ft.Container(height=10),
            ft.Text("🌙", size=60, text_align=ft.TextAlign.CENTER, selectable=True, font_family="Vazir"),
            ft.Text("روزچین | برنامه ریز شخصی", font_family="Vazir Bold", size=36, weight=ft.FontWeight.BOLD, color=c["text"], text_align=ft.TextAlign.CENTER, selectable=True),
            ft.Text(format_date_long(today), font_family="Vazir", size=18, color=c["text_sub"], text_align=ft.TextAlign.CENTER, selectable=True),
            ft.Container(height=10),
            ft.Text(progress_text, font_family="Vazir", size=16, color="#74B9FF", text_align=ft.TextAlign.CENTER, selectable=True) if progress_text else ft.Container(),
            ft.Container(height=20),
            make_button("افزودن روز جدید / ویرایش امروز", ft.Icons.ADD_CIRCLE_OUTLINE, new_day, "#A29BFE"),
            ft.Container(height=14),
            make_button("بررسی روزهای قبل", ft.Icons.HISTORY, show_history, "#74B9FF"),
            ft.Container(height=30),
        ], horizontal_alignment=ft.CrossAxisAlignment.CENTER, spacing=10, expand=True)

        show_view(view)

    def show_home(e=None):
        state["view"] = "home"
        state["current_date"] = None
        _render_home()

    # --- Editor ---
    def _render_editor():
        c = get_theme_colors(state["theme"])
        day = next((d for d in data["days"] if d["date"] == state["current_date"]), None)
        if not day:
            show_home()
            return

        def save_quote(val):
            day["quote"] = val
            save_data(data)

        def save_lesson(val):
            day["lesson"] = val
            save_data(data)

        quote_field = ft.TextField(value=day.get("quote", ""), hint_text="جمله‌ی امروزت چیه؟", multiline=True, min_lines=2, max_lines=4, text_style=ft.TextStyle(font_family="Vazir", size=17, color=c["text"]), hint_style=ft.TextStyle(font_family="Vazir", color=c["text_sub"]), border=ft.InputBorder.NONE, bgcolor="transparent", on_change=lambda e: save_quote(e.control.value))
        lesson_field = ft.TextField(value=day.get("lesson", ""), hint_text="درس امروزت چی بود؟", multiline=True, min_lines=3, max_lines=6, text_style=ft.TextStyle(font_family="Vazir", size=17, color=c["text"]), hint_style=ft.TextStyle(font_family="Vazir", color=c["text_sub"]), border=ft.InputBorder.NONE, bgcolor="transparent", on_change=lambda e: save_lesson(e.control.value))

        new_cat_name = ft.TextField(hint_text="نام دسته بندی (مثلا تکالیف عربی)...", text_style=ft.TextStyle(font_family="Vazir", size=16, color=c["text"]), hint_style=ft.TextStyle(font_family="Vazir", color=c["text_sub"]), border=ft.InputBorder.NONE, bgcolor=c["glass_border"], border_radius=10, expand=True)

        type_icons_row = ft.Row(spacing=10, wrap=True)

        def select_type(t):
            state["new_cat_type"] = t
            render_new_cat_icons()

        def render_new_cat_icons():
            type_icons_row.controls.clear()
            for t in CATEGORIES:
                is_sel = state["new_cat_type"] == t
                type_icons_row.controls.append(
                    ft.Container(
                        content=ft.Icon(CATEGORY_ICONS[t], color=CATEGORY_COLORS[t] if is_sel else c["text_sub"], size=24),
                        padding=10, border_radius=10,
                        bgcolor=f"#22{CATEGORY_COLORS[t][1:]}" if is_sel else c["glass_border"],
                        border=ft.Border.all(1, CATEGORY_COLORS[t] if is_sel else "transparent"),
                        on_click=lambda e, _t=t: select_type(_t)
                    )
                )
            page.update()

        def add_new_cat(e):
            name = new_cat_name.value.strip()
            if not name:
                return
            day["custom_categories"].append({"id": str(datetime.now().timestamp()), "name": name, "type": state["new_cat_type"], "tasks": [], "expanded": True})
            save_data(data)
            new_cat_name.value = ""
            render_categories()
            page.update()

        add_cat_box = glass(ft.Column([ft.Text("افزودن دسته بندی", font_family="Vazir Bold", size=16, color=c["text_sub"], selectable=True), ft.Row([new_cat_name, ft.IconButton(ft.Icons.ADD_CIRCLE, icon_color="#74B9FF", icon_size=32, on_click=add_new_cat)]), ft.Text("انتخاب آیکون برای دسته:", font_family="Vazir", size=14, color=c["text_sub"], selectable=True), type_icons_row], spacing=8), padding=16)

        categories_column = ft.Column(spacing=12)

        def render_categories():
            categories_column.controls.clear()
            for cat in day["custom_categories"]:
                categories_column.controls.append(render_category(cat))
            page.update()

        def render_category(cat):
            tasks = cat["tasks"]
            is_expanded = cat.get("expanded", True)
            color = CATEGORY_COLORS.get(cat["type"], "#74B9FF")
            icon = CATEGORY_ICONS.get(cat["type"], ft.Icons.CATEGORY)

            def toggle(e):
                cat["expanded"] = not cat["expanded"]
                save_data(data)
                render_categories()

            def delete_cat(e):
                day["custom_categories"] = [c_item for c_item in day["custom_categories"] if c_item["id"] != cat["id"]]
                save_data(data)
                render_categories()

            header = ft.Container(content=ft.Row([ft.Container(width=4, height=36, bgcolor=color, border_radius=2), ft.Icon(icon, color=color, size=24), ft.Text(cat["name"], font_family="Vazir Bold", size=18, weight=ft.FontWeight.BOLD, color=c["text"], expand=True, selectable=True), ft.Container(content=ft.Text(to_persian_digits(str(len(tasks))), font_family="Vazir Bold", size=14, color=color, selectable=True), padding=ft.Padding.symmetric(horizontal=10, vertical=2), border_radius=10, bgcolor=f"#22{color[1:]}"), ft.IconButton(ft.Icons.DELETE_OUTLINE, icon_color="#CC8888", icon_size=22, on_click=delete_cat), ft.Icon(ft.Icons.EXPAND_LESS if is_expanded else ft.Icons.EXPAND_MORE, color=c["text_sub"], size=24)], spacing=10), on_click=toggle)

            contents = [header]

            if is_expanded:
                task_list = ft.Column(spacing=6)
                for i, task in enumerate(tasks):
                    def make_handlers(idx):
                        def toggle_task(e):
                            cat["tasks"][idx]["done"] = not cat["tasks"][idx].get("done", False)
                            save_data(data)
                            render_categories()

                        def del_task(e):
                            cat["tasks"].pop(idx)
                            save_data(data)
                            render_categories()
                        return toggle_task, del_task

                    tt, dt = make_handlers(i)
                    task_list.controls.append(ft.Container(content=ft.Row([ft.Checkbox(value=task.get("done", False), on_change=tt), ft.Text(task["text"], font_family="Vazir", size=16, color=c["text"], expand=True, selectable=True, style=ft.TextStyle(decoration=ft.TextDecoration.LINE_THROUGH if task.get("done") else None)), ft.IconButton(ft.Icons.CLOSE, icon_size=18, icon_color="#CC8888", on_click=dt)]), padding=ft.Padding.symmetric(horizontal=10, vertical=4), border_radius=10, bgcolor=c["glass_border"]))

                if not tasks:
                    task_list.controls.append(ft.Text("هنوز کاری اضافه نکردی", font_family="Vazir", size=14, color=c["text_sub"], text_align=ft.TextAlign.CENTER, selectable=True))

    # --- History ---
    def _render_history():
        c = get_theme_colors(state["theme"])
        sorted_days = sorted(data["days"], key=lambda d: d["date"], reverse=True)

        if not sorted_days:
            items = [ft.Container(content=ft.Text("هنوز روزی ثبت نکردی 💤", font_family="Vazir", size=16, color=c["text_sub"], text_align=ft.TextAlign.CENTER, selectable=True), padding=60, alignment=ft.Alignment.CENTER)]
        else:
            items = []
            for d in sorted_days:
                def make_open(date):
                    def open_day(e):
                        state["current_date"] = date
                        show_editor()
                    return open_day

                total = sum(len(cat["tasks"]) for cat in d.get("custom_categories", []))
                done = sum(1 for cat in d.get("custom_categories", []) for t in cat["tasks"] if t.get("done"))
                quote = d.get("quote", "").strip() or "—"

                items.append(
                    ft.Container(
                        content=ft.Column([
                            ft.Row([
                                ft.Column([
                                    ft.Text(format_date_long(d["date"]), font_family="Vazir Bold", size=16, weight=ft.FontWeight.BOLD, color=c["text"], selectable=True),
                                    ft.Text(quote[:60] + ("..." if len(quote) > 60 else ""), font_family="Vazir", size=14, color=c["text_sub"], selectable=True)
                                ], spacing=4, expand=True),
                                ft.IconButton(ft.Icons.WIDGETS, icon_color="#00B894", icon_size=22, tooltip="ویجت این روز", on_click=lambda e, _d=d["date"]: launch_widget_for_date(_d)),
                                ft.Icon(ft.Icons.CHEVRON_LEFT, color=c["text_sub"], size=24)
                            ]),
                            ft.Container(height=8),
                            ft.Row([
                                ft.ProgressBar(value=(done / total) if total else 0, expand=True, color="#74B9FF", bgcolor=c["glass_border"], height=8),
                                ft.Text(to_persian_digits(f"{done}/{total}"), font_family="Vazir Bold", size=13, color="#74B9FF", selectable=True)
                            ], spacing=10)
                        ], spacing=4),
                        padding=ft.Padding.all(16),
                        border_radius=16,
                        bgcolor=c["glass_bg"],
                        border=ft.Border.all(1, c["glass_border"]),
                        on_click=make_open(d["date"]),
                        expand=True
                    )
                )

        view = ft.Column([
            ft.Row([ft.IconButton(ft.Icons.ARROW_BACK, on_click=show_home, icon_color=c["icon"], tooltip="بازگشت", icon_size=28), ft.Text("📚 روزهای قبل", font_family="Vazir Bold", size=22, weight=ft.FontWeight.BOLD, color=c["text"], expand=True, selectable=True), get_top_bar()], alignment=ft.MainAxisAlignment.SPACE_BETWEEN),
            ft.Column(items, spacing=12),
        ], spacing=12, expand=True)

        show_view(view)
